        // VERWERKEN VAN HET SAMENGESTELDE PROFIEL
        function handleProfileCreation(e) {
            e.preventDefault();

            const profileFor = document.querySelector('input[name="profileFor"]:checked').value;
            const system = document.getElementById('selected-system').value;
            let count = parseInt(document.getElementById('question-count').value) || 5;
            if(count > 10) count = 10;

            const scores = {};
            for(let i = 0; i < count; i++) {
                const val = document.getElementById(`dyn-q-${i}`).value;
                scores[val] = (scores[val] || 0) + 1;
            }

            // Bepaal de meest gekozen optie
            let highestOption = '';
            let maxCount = -1;
            for(const [key, cnt] of Object.entries(scores)) {
                if(cnt > maxCount) {
                    maxCount = cnt;
                    highestOption = key;
                }
            }

            let resultDisc = "DISC Rood (Direct)";
            let resultNlp = "Visueel (Zien)";

            if(system === 'DISC') {
                if(highestOption === 'geel') resultDisc = "DISC Geel (Relatie)";
                if(highestOption === 'groen') resultDisc = "DISC Groen (Ondersteunend)";
                if(highestOption === 'blauw') resultDisc = "DISC Blauw (Analytisch)";
                if(highestOption === 'rood') resultDisc = "DISC Rood (Direct)";
            } else {
                if(highestOption === 'auditief') resultNlp = "Auditief (Horen)";
                if(highestOption === 'kinesthetisch') resultNlp = "Kinesthetisch (Voelen)";
                if(highestOption === 'digitaal') resultNlp = "Auditief Digitaal (Logica)";
                if(highestOption === 'visueel') resultNlp = "Visueel (Zien)";
            }

            if(profileFor === 'self') {
                userProfile = system === 'DISC' ? { disc: resultDisc } : { nlp: resultNlp };
                addPoints(15);
                alert(`🎉 Jouw persoonlijke ${system}-profiel is berekend!\n\nResultaat: ${system === 'DISC' ? resultDisc : resultNlp}\n\n+15 Punten verdiend!`);

        function updateProgressBar() {
            if(deliverables.length === 0) {

            const percentage = Math.round((completedCount / deliverables.length) * 100);

            document.getElementById('progress-bar-fill').style.width = `${percentage}%`;
            document.getElementById('progress-percent-text').innerText = `${percentage}% Afgerond (${completedCount}/${deliverables.length})`;
        }

        function renderActiveKpis() {
            const container = document.getElementById('active-kpis-container');
            if(!container) return;
            container.innerHTML = '';

            const selectedKpis = kpiLibrary.filter(k => k.selected);

            if(selectedKpis.length === 0) {
                container.innerHTML = `<div class="col-span-3 text-center py-8 text-slate-400 font-bold text-xs">Geen actieve KPI's geselecteerd. Klik op "KPI's Beheren" om ze aan te vinken.</div>`;
                return;
            }

            selectedKpis.forEach(kpi => {
                const trendIcon = kpi.better === 'hoog' 
                    ? '<span class="text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded text-[10px] font-bold" title="Hoger is beter">📈 Hoger is beter</span>' 
                    : '<span class="text-amber-600 bg-amber-50 px-2 py-0.5 rounded text-[10px] font-bold" title="Lager is beter">📉 Lager is beter</span>';

                container.innerHTML += `
                    <div class="bg-white p-5 rounded-2xl border border-slate-200 shadow-sm space-y-3">
                        <div class="flex justify-between items-start">
                            <div>
                                <span class="text-[10px] font-bold text-emerald-700 uppercase block">${kpi.cat}</span>
                                <h3 class="font-black text-slate-900 text-sm mt-0.5">${kpi.abbr} - ${kpi.name}</h3>
                            </div>
                            <span class="text-xl font-black text-slate-900 bg-slate-100 px-3 py-1 rounded-xl">${kpi.val}</span>
                        </div>
                        <p class="text-xs text-slate-500 leading-relaxed">${kpi.desc}</p>
                        <div class="border-t pt-3 flex justify-between items-center text-xs">
                            ${trendIcon}
                            <button onclick="addPoints(10)" class="bg-emerald-100 hover:bg-emerald-200 text-emerald-800 font-bold px-2.5 py-1 rounded-lg text-[11px] transition-all">
                                +10 Ptn Claimen
                            </button>
                        </div>
                    </div>
                `;
            });
        }

        function renderKpiSelector() {

        function addNewKpi(e) {
            e.preventDefault();
            const cat = document.getElementById('new-kpi-cat').value;
            const abbr = document.getElementById('new-kpi-abbr').value;
            const val = document.getElementById('new-kpi-val').value;
            const desc = document.getElementById('new-kpi-desc').value;
            const trend = document.getElementById('new-kpi-trend').value;

            const newId = 'kpi_' + Date.now();

            kpiLibrary.push({
                id: newId,
                abbr: abbr,
                name: abbr,
                cat: cat,
                desc: desc,
                val: val,
                better: trend,
                selected: true
            });

            document.getElementById('new-kpi-abbr').value = '';
            document.getElementById('new-kpi-val').value = '';
            document.getElementById('new-kpi-desc').value = '';

            renderKpiSelector();
            renderActiveKpis();
            alert(`✅ KPI "${abbr}" succesvol toegevoegd en geactiveerd!`);
        }

        function saveKpiSelection() {
            kpiLibrary.forEach(kpi => {
                const chk = document.getElementById(`chk-${kpi.id}`);
                if(chk) kpi.selected = chk.checked;
            });
            renderActiveKpis();
            closeModal('configModal');
        }

        function addDeliverable(e) {
            e.preventDefault();
            deliverables.push({
                id: Date.now(),
                title: document.getElementById('del-title').value,
                freq: document.getElementById('del-freq').value,
                how: document.getElementById('del-how').value,
                holder: document.getElementById('del-holder').value,
                role: document.getElementById('del-role').value,
                profile: document.getElementById('del-profile').value,
                nlp: document.getElementById('del-nlp').value,
                koetjes: document.getElementById('del-koetjes').value,
                precheck: document.getElementById('del-precheck').value,
                deadline: document.getElementById('del-deadline').value,
                completed: false
            });
            closeModal('addDeliverableModal');
            renderDeliverablesTable();
            renderKoetjesCards();
            updateProgressBar();
        }

        function saveCoachingNote() {
            const note = document.getElementById('coaching-notes').value;
            if(!note) return alert('Voer een reflectie in.');
            addPoints(10);
            alert('Reflectie opgeslagen! +10 Punten verdiend.');

        }

        function handleLogin(e) {
            e.preventDefault();
            const name = document.getElementById('login-name').value;
            document.getElementById('user-display-name').innerText = name;
            document.getElementById('user-avatar').innerText = name.substring(0, 2).toUpperCase();
            document.getElementById('login-overlay').classList.add('hidden');
        }

        function logout() {
            if(confirm("Uitloggen?")) location.reload();
        }

        function switchTab(tab) {
            document.getElementById('tab-dashboard').classList.add('hidden');
            document.getElementById('tab-deliverables').classList.add('hidden');
            document.getElementById('tab-koetjes').classList.add('hidden');
            document.getElementById('tab-coaching').classList.add('hidden');
            
            const activeTab = document.getElementById('tab-' + tab);
            if(activeTab) activeTab.classList.remove('hidden');
        }

        function openModal(id) { 
            const m = document.getElementById(id);
            if(m) { m.classList.remove('hidden'); m.classList.add('flex'); }
        }

        function closeModal(id) { 
            const m = document.getElementById(id);
            if(m) { m.classList.add('hidden'); m.classList.remove('flex'); }
        }
